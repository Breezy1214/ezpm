    let mut body = String::new();
    ureq::get("https://api.github.com/repos/Breezy1214/ezpm/releases/latest")
        .header("Accept", "application/vnd.github.v3+json")
        .header("User-Agent", &format!("ezpm/{}", env!("CARGO_PKG_VERSION")))

        .body_mut()
        .read_to_string(&mut body)
        .ok()?;

    let json: serde_json::Value = serde_json::from_str(&body).ok()?;

Helper function:
```rust
fn print_version_footer(rx: &mpsc::Receiver<Option<String>>, current_ver: &str) {
    if let Ok(Some(latest)) = rx.recv_timeout(Duration::from_secs(2)) {
        if version::is_newer(current_ver, &latest) {
            eprintln!();
            eprintln!("Update available: v{} -> {} — run rokit update ezpm", current_ver, latest);
        }
    }
}
```

Note: Use `eprintln` for the version footer so it goes to stderr (does not interfere with stdout piping). The version footer uses a 2-second timeout so it never blocks the main thread for long (Pitfall 6).

3. **Menu loop:**
```rust
pub fn run_interactive_menu() {
    let version = env!("CARGO_PKG_VERSION");

    loop {
        print_logo(version);

        let labels: Vec<&str> = MENU_ITEMS.iter().map(|(label, _)| *label).collect();

        let result = inquire::Select::new("What would you like to do?", labels).prompt();

        match result {
            Ok(selection) => {
                // Find the command key for the selected label
                let cmd = MENU_ITEMS.iter()
                    .find(|(label, _)| *label == selection)
                    .map(|(_, cmd)| *cmd)
                    .unwrap_or("");

                // If user selected a header (empty cmd), loop back (Pitfall 1)
                if cmd.is_empty() {
                    continue;
                }

                if cmd == "exit" {
                    println!("Goodbye!");
                    std::process::exit(0);
                }

                // Execute the selected command
                let result = run_command(cmd);
                if let Err(e) = result {
                    eprintln!("Error: {}", e);
                }

                // After command completes, loop back to menu
                println!();
            }
            Err(_) => {
                // User pressed Ctrl-C or Escape
                std::process::exit(0);
            }
        }
    }
}
```

4. **`run_command(cmd: &str) -> anyhow::Result<()>`** — dispatch function:
This function loads config fresh and dispatches to the right handler based on the command key string. It needs to:
- Load config for each command invocation (so alias changes are picked up).
- Match on command key strings and call the appropriate handler.
- For "serve": print "serve is coming in a future update."
- For fix-requires: call the require_fixer with loaded config.

```rust
fn run_command(cmd: &str) -> anyhow::Result<()> {
    // Load fresh config for each command
    let (cfg, warnings) = crate::config::load_config()?;
    for w in &warnings { eprintln!("{}", w); }

    let src = cfg.paths.as_ref()
        .and_then(|p| p.src.as_deref())
        .unwrap_or("src");

    match cmd {
        "init" => crate::commands::init::run_init(),
        "install" => crate::commands::install::install_tools(src),
        "setup-wally-packages" => crate::commands::install::setup_wally_packages(src),
        "alias-add" => crate::commands::alias::alias_add(),
        "alias-remove" => crate::commands::alias::alias_remove(),
        "alias-list" => crate::commands::alias::alias_list(&cfg.aliases),
        "alias-sync" => crate::commands::alias::alias_sync(),
        "lint" => crate::commands::quality::lint(src),
        "format" => crate::commands::quality::format_code(src),
        "docs" => {
            let docs_enabled = cfg.display.as_ref()
                .and_then(|d| d.docs_enabled)
                .unwrap_or(false);
            crate::commands::quality::docs(docs_enabled)
        }
        "fix-requires" => {
            let aliases = cfg.aliases.unwrap_or_default();
            let root_dir = std::env::current_dir()?;
            let result = crate::services::require_fixer::fix_requires(&root_dir, &aliases, src)?;
            if result.files_changed == 0 {
                println!("All requires up to date. 0 changes across {} files.", result.total_files_scanned);
            } else {
                let total_rewrites: usize = result.changes.iter().map(|c| c.rewrites.len()).sum();
                for file_change in &result.changes {
                    println!("{}:", file_change.file.display());
                    for rw in &file_change.rewrites {
                        println!("  {} -> {}", rw.old, rw.new);
                    }
                }
                println!();
                println!("Fixed {} requires across {} files", total_rewrites, result.files_changed);
            }
            Ok(())
        }
        "serve" => {
            println!("serve is coming in a future update. Current version: {}", env!("CARGO_PKG_VERSION"));
            Ok(())
        }
        _ => {
            eprintln!("Unknown command: {}", cmd);
            Ok(())
        }
    }
}
```
